{--------------------------------------------}
{ 公式一：趋势共振选股（条件选股·CTRL+T）      }
{ 逻辑：60日线向上 + 股价站上60日线            }
{       + MACD 0轴上方金叉 + RSI<80            }
{--------------------------------------------}
MA60:=MA(CLOSE,60);
MA20:=MA(CLOSE,20);
DIFF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIFF,9);
MACD2:=(DIFF-DEA)*2;
RSI14:=SMA(MAX(CLOSE-REF(CLOSE,1),0),14,1)/SMA(ABS(CLOSE-REF(CLOSE,1)),14,1)*100;

趋势合格:=MA60>REF(MA60,5) AND CLOSE>MA60;
节奏共振:=DIFF>0 AND CROSS(DIFF,DEA) OR (DIFF>DEA AND MACD2>REF(MACD2,1) AND REF(MACD2,1)>REF(MACD2,2));
量能配合:=VOL>REF(VOL,1)*0.8 AND VOL<REF(VOL,1)*2.5;

选股:趋势合格 AND 节奏共振 AND RSI14<80 AND 量能配合;
